\begin{frame}{1. True Embedding with LSBM and Hamming Codes}
    \textbf{Approach}
    \begin{itemize}
        \item Implemented matrix embedding explicitly by generating parity-check matrices $H$ for Hamming codes.
        \item \textbf{Syndrome Calculation}: Compute $s = H \cdot y \pmod{2}$. The required change is $m - s \pmod{2}$.
        \item \textbf{Modification}: Find the index to change based on the difference. 
        \item \textbf{LSBM}: In LSB Matching, changing the LSB equates to randomly adding or subtracting $1$ from the pixel intensity, ensuring boundaries (like $0$ and $x_{max}$) are respected.
    \end{itemize}
    \vspace{0.5em}
    \textbf{Advantages}
    \begin{itemize}
        \item Significantly increases embedding efficiency $e$ compared to baseline steganography.
        \item Reduces the number of noticeable modifications (changes) in the cover object.
    \end{itemize}
\end{frame}

\begin{frame}{2. Simulating Hamming Codes}
    \textbf{Simulation Strategy}
    \begin{itemize}
        \item Each Hamming code block changes with probability $P(\text{change}) = \frac{2^k - 1}{2^k}$.
        \item The total number of modified blocks can be sampled directly using a Binomial distribution: \\ $\sim \text{Binomial}\left(N_{\text{blocks}}, \frac{2^k - 1}{2^k}\right)$.
        \item Solves the need to explicitly execute code matrices over covers, massively increasing simulation speed.
    \end{itemize}
    \vspace{0.5em}
    \textbf{Interpolation for Target $\alpha$}
    \begin{itemize}
        \item To achieve non-integer embedding rates $\alpha$, we linearly interpolate the change probabilities per element of the two neighboring Hamming codes $k$ and $k+1$.
    \end{itemize}
\end{frame}

\begin{frame}{3. Performance and Precision Evaluation}
    \textbf{Speed and Precision (Simulation vs. True Embedding)}
    \begin{itemize}
        \item \textbf{Precision:} The simulated embedding efficiency $e_{\text{sim}}$ matches the $e_{\text{true}}$ of fully processed image blocks up to high statistical precision.
        \item \textbf{Speedup:} The simulation reduces evaluation times from the millisecond/second range down to magnitudes of microseconds ($>10^3\times$ speedup).
    \end{itemize}
    \vspace{0.5em}
    \textbf{Detectability vs. Uncoded LSBM}
    \begin{itemize}
        \item \textbf{Uncoded LSBM Baseline:} Random message flipping results in an embedding efficiency of $e \approx 2$ (bits embedded per change).
        \item \textbf{Coded Integrity:} The calculated analytical $e$ progressively climbs well above $2$, scaling directly with block sizes (reducing $\alpha$).
        \item \textbf{Detectability Gain:} Because detectability correlates directly with the number of changes, our steganography achieves a significant drop in detectability (often reducing changes by multiples).
    \end{itemize}
\end{frame}
